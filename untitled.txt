<script>
let currentSubject = 'Physics';

// Default initial data
const defaultLectures = {
  Physics: [{ title: 'Electric Charges and Fields', videoId: '_oMIg50JLI8' }],
  Chemistry: [{ title: 'Structure of Atom', videoId: '_oMIg50JLI8' }],
  Biology: [{ title: 'Cell Cycle & Division', videoId: '_oMIg50JLI8' }]
};

// Load saved data from browser storage
let lecturesData = JSON.parse(localStorage.getItem('neet_classroom_data')) || defaultLectures;

function saveData() {
  localStorage.setItem('neet_classroom_data', JSON.stringify(lecturesData));
}

function extractVideoID(url) {
  const regExp = /^.*(youtu.be\/|v\/|u\/\w\/|embed\/|watch\?v=|\&v=)([^#\&\?]*).*/;
  const match = url.match(regExp);
  return (match && match[2].length === 11) ? match[2] : url;
}

function addNewLecture() {
  const subject = document.getElementById('sub-select').value;
  const title = document.getElementById('chapter-title').value.trim();
  const rawUrl = document.getElementById('youtube-url').value.trim();

  if(!title || !rawUrl) {
    alert("Kripya Title aur YouTube Link dono bharein!");
    return;
  }

  const videoId = extractVideoID(rawUrl);
  
  if (!lecturesData[subject]) {
    lecturesData[subject] = [];
  }

  lecturesData[subject].push({ title: title, videoId: videoId });
  saveData();

  document.getElementById('chapter-title').value = '';
  document.getElementById('youtube-url').value = '';

  alert('Lecture successfully add ho gaya hai!');
  show(subject);
}

function deleteLecture(index) {
  if (confirm("Kya aap is lecture ko delete karna chahte hain?")) {
    lecturesData[currentSubject].splice(index, 1);
    saveData();
    show(currentSubject);
  }
}

function show(subject) {
  currentSubject = subject;
  const listContainer = document.getElementById('list');
  const items = lecturesData[subject] || [];
  
  if (items.length === 0) {
    listContainer.innerHTML = '<p style="color:#aaa;">Is subject mein abhi koi lecture add nahi hai.</p>';
    return;
  }

  listContainer.innerHTML = items.map((item, index) => `
    <div class="card">
      <div class="card-info">
        <span class="tag">${subject}</span>
        <h3 style="margin:5px 0;">${item.title}</h3>
        <p style="margin:0;color:#aaa;font-size:14px;">Lecture ${index + 1}</p>
      </div>
      <div>
        <button class="play-btn" onclick="playVideo('${item.videoId}', '${item.title}')">▶ Play</button>
        <button class="delete-btn" onclick="deleteLecture(${index})">🗑</button>
      </div>
    </div>
  `).join('');
}

function playVideo(videoId, title) {
  const screen = document.getElementById('video-screen');
  const nowPlaying = document.getElementById('now-playing');
  
  screen.innerHTML = `<iframe src="https://www.youtube.com/embed/${videoId}?autoplay=1" allow="autoplay; encrypted-media" allowfullscreen></iframe>`;
  nowPlaying.innerText = `Currently Playing: ${title}`;
  
  window.scrollTo({ top: 0, behavior: 'smooth' });
}

show('Physics');
